$ starship init zsh | head -6
# ZSH has a quirk where `preexec` is only run if a command is actually run (i.e
# pressing ENTER at an empty command line will not cause preexec to fire). This
# can cause timing issues, as a user who presses "ENTER" without running a command
# will see the time to the start of the last command, which may be very large.

# To fix this, we create STARSHIP_START_TIME upon preexec() firing, and destroy it

$ starship init fish | head -6
source (/home/hatch/workspace/build-check/starship-scratch/starship init fish --print-full-init | psub)
$ starship init powershell | head -6
Invoke-Expression (& '/home/hatch/workspace/build-check/starship-scratch/starship' init powershell --print-full-init | Out-String)
